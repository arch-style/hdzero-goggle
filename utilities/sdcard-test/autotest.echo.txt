# 再開後の音（エコー、音が消える）の確認。再開のたびに音声デコーダを作り直す版。
# mp4: 8 秒そのまま、そのあと「止める→コマ送り→再開」を 5 回（各 3 秒）。
# TS: 3 秒そのまま、同じく 5 回。
# 音の状態は 1 秒ごとの "VS audio" 行、再開ごとの "VS audio play N ... new decoder"。
# デコーダが止まれば "has given nothing"、音声フレームの飛びは "does not follow"。
size frame
stepgap 400

echo mp4
open hdz_9003.mp4
wait 8000
loop 5
  pause
  wait 500
  step +1
  play
  waitresume 5000
  wait 3000
end
exit
wait 1500

echo ts
open hdz_9048.ts
wait 3000
loop 5
  pause
  wait 500
  step +1
  play
  waitresume 5000
  wait 3000
end
exit
