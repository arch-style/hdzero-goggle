# 写真は不要です（約 3 分）。取り込み画像と、その中の「透けた画素」(明るさ 255) の数で調べます。
#  1) ディスプレイの画質設定 (enhance_*, color_temperature) が出力に効くか
#  2) 再生側で 1 コマずつ明るさ 125 を書き換えたときの手間 (時間) と、透けた画素が減るか
#     (1080p60 と、新しく試す 720p60 のライブ側)
# 途中で画面の色や明るさが変わったり、止まったり動いたりします。
size frame
shell for a in enhance_mode enhance_bright enhance_contrast enhance_saturation enhance_edge enhance_detail enhance_denoise color_temperature; do echo $a=$(cat /sys/class/disp/disp/attr/$a); cp /sys/class/disp/disp/attr/$a /tmp/$a.orig; done
pbdisplay 60
open /mnt/extsd/keytest.ts
wait 1500
pause
wait 500
label E0 as is
grab k60
label E1 enhance_mode 1
shell echo 1 > /sys/class/disp/disp/attr/enhance_mode; echo enhance_mode now $(cat /sys/class/disp/disp/attr/enhance_mode)
wait 700
grab k_e1
shell cat /tmp/enhance_mode.orig > /sys/class/disp/disp/attr/enhance_mode; echo enhance_mode back $(cat /sys/class/disp/disp/attr/enhance_mode)
label E2 enhance_mode 2
shell echo 2 > /sys/class/disp/disp/attr/enhance_mode; echo enhance_mode now $(cat /sys/class/disp/disp/attr/enhance_mode)
wait 700
grab k_e2
shell cat /tmp/enhance_mode.orig > /sys/class/disp/disp/attr/enhance_mode; echo enhance_mode back $(cat /sys/class/disp/disp/attr/enhance_mode)
label E3 enhance_bright 80
shell echo 80 > /sys/class/disp/disp/attr/enhance_bright; echo enhance_bright now $(cat /sys/class/disp/disp/attr/enhance_bright)
wait 700
grab k_e3
shell cat /tmp/enhance_bright.orig > /sys/class/disp/disp/attr/enhance_bright; echo enhance_bright back $(cat /sys/class/disp/disp/attr/enhance_bright)
label E4 enhance_contrast 80
shell echo 80 > /sys/class/disp/disp/attr/enhance_contrast; echo enhance_contrast now $(cat /sys/class/disp/disp/attr/enhance_contrast)
wait 700
grab k_e4
shell cat /tmp/enhance_contrast.orig > /sys/class/disp/disp/attr/enhance_contrast; echo enhance_contrast back $(cat /sys/class/disp/disp/attr/enhance_contrast)
label E5 enhance_saturation 80
shell echo 80 > /sys/class/disp/disp/attr/enhance_saturation; echo enhance_saturation now $(cat /sys/class/disp/disp/attr/enhance_saturation)
wait 700
grab k_e5
shell cat /tmp/enhance_saturation.orig > /sys/class/disp/disp/attr/enhance_saturation; echo enhance_saturation back $(cat /sys/class/disp/disp/attr/enhance_saturation)
label E6 color_temperature 100
shell echo 100 > /sys/class/disp/disp/attr/color_temperature; echo color_temperature now $(cat /sys/class/disp/disp/attr/color_temperature)
wait 700
grab k_e6
shell cat /tmp/color_temperature.orig > /sys/class/disp/disp/attr/color_temperature; echo color_temperature back $(cat /sys/class/disp/disp/attr/color_temperature)
label E7 color_temperature -100
shell echo -100 > /sys/class/disp/disp/attr/color_temperature; echo color_temperature now $(cat /sys/class/disp/disp/attr/color_temperature)
wait 700
grab k_e7
shell cat /tmp/color_temperature.orig > /sys/class/disp/disp/attr/color_temperature; echo color_temperature back $(cat /sys/class/disp/disp/attr/color_temperature)
wait 500
grab k60b
exit
wait 1000
